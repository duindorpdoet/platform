test("eigen rustige weergave blijft na herladen en navigatie bewaard", async ({ page }) => {
  await page.emulateMedia({ reducedMotion: "no-preference" });
  await page.goto("/");
  const pause = page.getByRole("button", { name: "Pauzeer sfeereffecten", exact: true });
  await pause.focus(); await page.keyboard.press("Enter");
  await expect(page.locator(".theme-poster")).toHaveAttribute("data-motion", "still");
  expect(await page.evaluate(() => localStorage.getItem("frightnight-calm-v1"))).toBe("1");
  await page.reload();
  await expect(page.getByRole("button", { name: "Speel sfeereffecten", exact: true })).toHaveAttribute("aria-pressed", "true");
  await page.getByRole("link", { name: "Mijn Nachtpas", exact: true }).click();
  await expect(page.locator(".fn-calm-toggle")).toHaveAttribute("aria-pressed", "true");
  await expect(page.locator("html")).toHaveClass(/fn-calm/);
  await page.getByRole("link", { name: "Naar het festival", exact: true }).click();
  await expect(page.locator(".theme-poster")).toHaveAttribute("data-motion", "still");
  await page.goto("/mijn-zone");
  const resume = page.getByRole("button", { name: "Sfeer hervatten", exact: true });
  await resume.focus(); await page.keyboard.press("Space");
  await expect(page.locator(".fn-calm-toggle")).toHaveAttribute("aria-pressed", "false");
  await page.getByRole("link", { name: "Naar het festival", exact: true }).click();
  await expect(page.locator(".theme-poster")).toHaveAttribute("data-motion", "full");
  await page.reload();
  await expect(page.getByRole("button", { name: "Pauzeer sfeereffecten", exact: true })).toHaveAttribute("aria-pressed", "false");
});

test("de gepubliceerde tekstprojectie vult programma en voorwaarden afzonderlijk aan", async ({ page }) => {
  // Presentation contract only: this endpoint fixture represents rows already
  // published by the server. Database publication/edition rules have SQL tests.
  await page.route("**/api/frightnight/public", route => route.fulfill({ json: { data: { content: [
    { key: "programme", body: "Programma-aanvulling uit de gepubliceerde projectie.", version: 2 },
    { key: "terms", body: "Voorwaardenaanvulling uit de gepubliceerde projectie.", version: 3 },
  ], stale: false } } }));
  await page.goto("/voorwaarden");
  const terms = page.getByRole("complementary", { name: "Gepubliceerde voorwaardenaanvulling" });
  await expect(terms).toContainText("Voorwaardenaanvulling uit de gepubliceerde projectie.");
  await expect(page.locator("body")).not.toContainText("Programma-aanvulling uit de gepubliceerde projectie.");
  await expect(page.locator("main")).toContainText("17:00 tot 18:30");
  await expect(page.locator("main")).toContainText("18:30 tot 21:00");
  await expect(page.locator("main")).toContainText("Versie: 2027-test-v1");
  await page.goto("/");
  await expect(page.getByRole("complementary", { name: "Gepubliceerde informatie van de organisatie" })).toContainText("Programma-aanvulling uit de gepubliceerde projectie.");
  await expect(page.locator("body")).not.toContainText("Voorwaardenaanvulling uit de gepubliceerde projectie.");
  await expect(page.locator(".pp-programme-strip")).toContainText("17:00 — 18:30");
});

test("productieserviceworker wordt actief en levert offline noodinformatie met herstel", async ({ browser, baseURL }) => {
  const context = await browser.newContext({ serviceWorkers: "allow", viewport: { width: 390, height: 844 }, reducedMotion: "reduce" });
  const origin = new URL(baseURL!).origin;
  const forbidden: string[] = [];
  await context.route("**/*", route => {
    const request = route.request(); const url = new URL(request.url());
    if (!["GET", "HEAD", "OPTIONS"].includes(request.method()) || url.origin !== origin || /^\/auth\/|^\/api\/(?:frightnight\/(?:command|edition|push|logout)|mail|push|payments?)(?:\/|$)/.test(url.pathname)) {
      forbidden.push(`${request.method()} ${url.origin}${url.pathname}`);
      return route.abort("blockedbyclient");
    }
    return route.continue();
  });
  const page = await context.newPage();
  try {
    // A static page lets this browser prepare harmless obsolete-cache fixtures
    // before the production layout registers its real service worker.
    await page.goto(`${origin}/offline.html`);
    await page.evaluate(async () => {
      const legacy = await caches.open("duindorp-public-browser-regression-obsolete");
      await legacy.put("/obsolete-public-fixture", new Response("obsolete public fixture"));
      const privateCache = await caches.open("duindorp-private-browser-regression-obsolete");
      await privateCache.put("/nachtpas", new Response("cached private fixture"));
    });
    await page.goto(baseURL!);
    await page.waitForFunction(async () => {
      const registration = await navigator.serviceWorker.ready;
      return registration.active?.state === "activated" && navigator.serviceWorker.controller?.state === "activated";
    }, undefined, { timeout: 20_000 });
    // Activating after an obsolete public cache also refreshes the open page.
    // Let that real navigation settle before inspecting its new document.
    await page.waitForLoadState("networkidle");
    expect(await page.evaluate(() => navigator.serviceWorker.controller?.scriptURL)).toBe(`${origin}/sw.js`);
    expect(await page.evaluate(async () => (await navigator.serviceWorker.ready).pushManager.getSubscription())).toBeNull();
    const keys = await page.evaluate(() => caches.keys());
    expect(keys).not.toContain("duindorp-public-browser-regression-obsolete");
    expect(keys.some(key => key.startsWith("duindorp-private"))).toBe(false);
    await context.setOffline(true);
    await page.reload({ waitUntil: "domcontentloaded" });
    await expect(page.getByRole("heading", { level: 1 })).toContainText("EVEN GEEN VERBINDING");
    await expect(page.locator("body")).toContainText("Duinpad / Oogstdorp");
    await expect(page.locator("body")).toContainText("lokale bediening");
    // A precached asset must actually be served when the network is absent.
    // Merely finding its cache entry does not prove that the offline page works.
    await expect.poll(() => page.locator("main img").evaluate(image => {
      const logo = image as HTMLImageElement;
      return logo.complete && logo.naturalWidth > 0;
    }), { timeout: 5_000 }).toBe(true);
    await page.locator("main img").evaluate(image => (image as HTMLImageElement).decode());
    const cached = await page.evaluate(async () => {
      const paths: string[] = [];
      for (const key of await caches.keys()) {
        const cache = await caches.open(key);
        for (const request of await cache.keys()) paths.push(new URL(request.url).pathname);
      }
      return paths;
    });
    expect(cached).toContain("/offline.html");
    expect(cached.filter(path => path === "/frightnight/icon-192.png")).toHaveLength(1);
    expect(cached.some(path => /^\/(?:api|auth|inloggen|nachtpas|avondregie|testcentrum)(?:\/|$)|^\/mijn-/.test(path))).toBe(false);
    const privateResponse = await page.goto(`${origin}/nachtpas`, { waitUntil: "domcontentloaded" });
    expect(privateResponse?.status()).toBe(503);
    expect(privateResponse?.headers()["cache-control"]).toContain("private, no-store");
    await expect(page.getByRole("heading", { level: 1 })).toContainText("EVEN GEEN VERBINDING");
    await expect(page.locator("body")).not.toContainText("cached private fixture");
    await expect.poll(() => page.locator("main img").evaluate(image => (image as HTMLImageElement).naturalWidth > 0), { timeout: 5_000 }).toBe(true);
    await context.setOffline(false);
    await page.goto(baseURL!);
    await expect(page.getByRole("heading", { level: 1 })).toHaveText("DIT WORDTJOUW NACHT.");
    expect(forbidden).toEqual([]);
  } finally {
    await context.setOffline(false);
    await context.close();
  }
});
